enum { IDLE, FALL_DETECTED, BUILD, SEND, WAIT_ACK, DONE, FAILED };
#define G_THRESH 2.5f
#define CANCEL_MS 8000
#define ACK_TO_MS 3000
#define MAX_RETRY 3

void main(void){
  hw_init(); sensors_init(); state = IDLE;
  for(;;){
    switch(state){
    case IDLE:
      if(sos_pressed()) { type=SOS; state=BUILD; }
      else if(fall_detect()) { type=FALL; t0=millis(); alert_on(); state=FALL_DETECTED; }
      break;
    case FALL_DETECTED:
      if(cancel_pressed()) { alert_off(); state=IDLE; }
      else if(millis()-t0 > CANCEL_MS) state=BUILD;
      break;
    case BUILD:
      seq++; pkt = make_packet(type, seq, gps_or_zero());
      aes128_encrypt(key_enc, pkt.payload);
      aes_cmac(key_mac, &pkt, pkt.tag);
      retry=0; state=SEND; break;
    case SEND: radio_tx(&pkt); t1=millis(); state=WAIT_ACK; break;
    case WAIT_ACK:
      if(radio_rx(&ack) && ack.seq==seq && cmac_ok(&ack)) state=DONE;
      else if(millis()-t1 > ACK_TO_MS) state = (++retry < MAX_RETRY) ? SEND : FAILED;
      break;
    case DONE: case FAILED: alert_off(); state=IDLE; break;
    }
  }
}
